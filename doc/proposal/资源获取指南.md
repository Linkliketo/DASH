du -sh ~/data/<dataset> && find ~/data/<dataset> -type f | wc -l

## 五、云实例目录结构（建议布局）

```
~/
├── repos/                  # 第三方代码（§二），不进 DASH 仓库
├── dash/                   # DASH 自有代码（git 管理，GitHub 私有→开源）
├── assets/                 # MPI 资产（§3.2，禁止外发）
│   ├── smplx/  flame/  smpl/
├── autodl-tmp/             # 数据盘（免费 50GB，高 IO）
│   └── data/               # 或软链接 ~/data → 此处
│       ├── express4d/  selfcap/  3dpw/  casme2/  disfa/ ...
├── autodl-fs/              # 文件存储（免费 20GB，跨实例共享）
│   └── checkpoints/        # 训练产出权重备份
└── autodl-nas/             # 网盘（上传下载中转）
```

```bash
mkdir -p ~/repos ~/dash ~/assets/{smplx,flame,smpl}
ln -s /root/autodl-tmp/data ~/data
```
